byte[,] fft_shift(byte[,] f)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];

   for (int y=0;y<h/2;y++)
      for (int x=0;x<w/2;x++)
      {
          g[x+w/2,y+h/2] = f[x,y];
          g[x,y] = f[x+w/2,y+h/2];

          g[x+w/2,y] = f[x,y+h/2];
          g[x,y+h/2] = f[x+w/2,y];
      } 
   return g;
}

void main()
{
   byte[,] f = LoadImg();
   ShowImg("f",f);

   byte[,] g = fft_shift(f);
   ShowImg("g",g);
}
