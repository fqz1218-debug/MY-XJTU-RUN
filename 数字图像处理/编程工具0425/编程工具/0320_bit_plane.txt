// n:[0..7]

byte[,] get_bit_plane(byte[,] f,int n)  
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];

   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
       // if ((f[x,y]&(1<<n))!=0) g[x,y]=255;
       g[x,y] = (byte)(((f[x,y]>>n)&1)*255);
        
   return g;
}

void main()
{
   byte[,] f = LoadImg();
   ShowImg("f",f);

  for (int n=0;n<8;n++)
    ShowImg("bitplane"+n.ToString(),get_bit_plane(f,n));

}
