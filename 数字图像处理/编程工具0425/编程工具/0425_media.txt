
void add_noise(byte[,] f,int s_count,int p_count)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   for (int n=0;n<s_count;n++)
   {
      int x = (int)(Rand()*w);
      int y = (int)(Rand()*h);

      f[x,y] = 255;
   }

   for (int n=0;n<p_count;n++)
   {
      int x = (int)(Rand()*w);
      int y = (int)(Rand()*h);

      f[x,y] = 0;
   }
}

void order(byte[] a)
{
   int len = a.Length;

   for (int i=0;i<len-1;i++)
      for (int j=i+1;j<len;j++)  
          if (a[i]>a[j])
          {
             byte temp = a[i];
             a[i] = a[j];
             a[j] = temp; 
          }  
 
}

byte media(byte[] a)
{
   order(a);

   return a[a.Length/2];
}

byte[,] media_filter(byte[,] f,int M,int N)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];

   byte[] a = new byte[1+2*M+2*N];

   for (int y=N;y<h-N;y++)
     for (int x=M;x<w-M;x++)
     {
        int i = 0; 
        a[i++] = f[x,y];
        
        for (int j=1;j<=M;j++) { a[i++] = f[x+j,y]; a[i++] = f[x-j,y]; }   
        for (int j=1;j<=N;j++) { a[i++] = f[x,y+j]; a[i++] = f[x,y-j]; }   
        
        

        g[x,y] = media(a); 
     }

   return g;
}
