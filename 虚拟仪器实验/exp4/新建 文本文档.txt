fs = 10000;
dt = 1/fs;
zerocross = [];

for i = 1:length(signal)-1
    if signal(i)*signal(i+1) < 0
        x = signal(i) / (signal(i) - signal(i+1));
        zerocross = [zerocross, (i + x)*dt];
    end
end

if length(zerocross) >= 2
    T = 2*(zerocross(end)-zerocross(1))/(length(zerocross)-1);
    freq = 1/T;
else
    freq = 0;
end




fs = 10000;
N = length(signal);
s = 0; c = 0;

for i = 2:N-1
    u1 = signal(i-1);
    u2 = signal(i);
    u3 = signal(i+1);

    temp = (u1 + u3)/(2*u2);
    w = acos(temp);

    s = s + w;
    c = c + 1;
end

if c > 0
    w_avg = s/c;
    freq = fs*w_avg/(2*pi);
else
    freq = 0;
end




fs = 10000;
N = length(signal);
s = 0; c = 0;
A = max(abs(signal));

for i = 2:N-1
    u1 = signal(i-1);
    u2 = signal(i);
    u3 = signal(i+1);

    if abs(u2) > 0.7*A || abs(u2) < 0.3*A
        continue;
    end

    temp = (u1 + u3)/(2*u2);

    if abs(temp) > 0.9
        continue;
    end

    w = acos(temp);
    s = s + w;
    c = c + 1;
end

if c > 0
    w_avg = s/c;
    freq = fs*w_avg/(2*pi);
else
    freq = 0;
end
