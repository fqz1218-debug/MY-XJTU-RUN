int get_m1(byte[,] f,int thr)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   int sum = 0;
   int count = 0;
   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
     {
        if (f[x,y]>thr) 
        {
           sum+=f[x,y];
           count++;

        }  
     }

   if (count!=0) return sum/count;
   return -1; 
}

int get_m2(byte[,] f,int thr)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   int sum = 0;
   int count = 0;
   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
     {
        if (f[x,y]<=thr) 
        {
           sum+=f[x,y];
           count++;

        }  
     }

   if (count!=0) return sum/count;
   return -1; 

}


int get_thr(byte[,] f,int thr0)
{
   while(true)
   {
      int m1 = get_m1(f,thr0);
      int m2 = get_m2(f,thr0);

      int thr1 = (m1+m2)/2;

      print(thr1);

      if (Abs(thr1-thr0)<5) return thr1;

      thr0 = thr1;
   }

}


byte[,] img_seg(byte[,] f,int thr)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];

   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
        if (f[x,y]>thr) g[x,y]=255;
        else g[x,y] = 0; 

   return g;
}

int get_mean(byte[,] f)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   int sum = 0;
   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
     {
           sum+=f[x,y];
     }

   return sum/(w*h);
}
