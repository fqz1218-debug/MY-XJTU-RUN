byte[,] add_noise(byte[,] f,int level)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];

   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
        g[x,y]= S(f[x,y]+((Rand()-0.5)*2*level));
         
   return g;
}

byte S(double v)
{
   if (v<0) return 0;
   if (v>255) return 255;
   return (byte)v;
}

void main()
{
   byte[,] f = LoadImg();
   ShowImg("f",f);

   ShowImg("f+noise(40)",add_noise(f,40));
}
