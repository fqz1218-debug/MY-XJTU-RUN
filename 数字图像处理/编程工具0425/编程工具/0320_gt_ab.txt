byte[,] gt_ab(byte[,] f,double a,double b)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];


   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
        g[x,y]=S(f[x,y]*a+b);

   return g;
}

byte S(double v)
{
   if (v<0) return 0;
   if (v>255) return 255;
   return (byte)v;
}


void main()
{
   byte[,] f = LoadImg();
   ShowImg("f",f);

  byte[,] f1 = gt_ab(f,0.5,0);
  byte[,] f2 = gt_ab(f,0.5,128);
  byte[,] f3 = gt_ab(f,0.5,64);

  ShowImg("f1",f1);
  ShowImg("f2",f2);
  ShowImg("f3",f3);

}
