
byte[,] get_n_bits_img(byte[,] f,int n)
{
   int w = f.GetLength(0);
   int h = f.GetLength(1);

   byte[,] g = new byte[w,h];


   int m = 8-n;
   for (int y=0;y<h;y++)
     for (int x=0;x<w;x++)
     {
         g[x,y] = (byte)((f[x,y]>>m)<<m);
     }

   return g;
}
